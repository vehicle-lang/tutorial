# Vehicle itself. Deliberately not pinned to one version: Chapter 1 tells
# readers to run `pip install vehicle-lang`, so CI should test against whatever
# that gives them today. Pinning here is how this file drifted to 0.17.0 while
# the chapters documented a much newer command-line interface.
#
# The lower bound is real, though: Chapter 4's property-driven training does
# not work below 0.28.0, because the adversarial search inside the compiled
# loss stepped in the wrong direction (fixed in 0.28.0). The scripts run
# without error on 0.27.1 and simply train networks that are less robust.
vehicle-lang>=0.28.0

# The neural network verifier used from Chapter 2 onwards. Provides the
# `Marabou` executable that `vehicle verify --solver Marabou` looks for.
maraboupy   ==2.0.0

# PyTorch route (chapter-4/chapter-code/pt_classifier.py). Unpinned, but note
# that `torch` and `numpy` must be ABI-compatible: a torch built against
# NumPy 1.x fails under NumPy 2.x with `RuntimeError: Numpy is not available`.
# Verified working with torch 2.13.0 and numpy 2.2.6.
torch
torchvision

# Required by `torch.onnx.export` in recent PyTorch, which routes through a new
# exporter that imports onnxscript. Without it, training completes and only the
# final export step fails with ModuleNotFoundError.
onnxscript  ==0.7.1

# TensorFlow route (chapter-4/chapter-code/tf_classifier.py). tf2onnx converts
# the exported SavedModel to ONNX. Not tested on the maintainers' machine.
tensorflow
tf2onnx
